cmake_minimum_required(VERSION 3.21)

# Keep the retained CB77 implementation testable in a separate build tree.
option(COOP_REFERENCE_ONLY "Build retained v0.0.37/CB77 checks instead of session foundation" OFF)
if(COOP_REFERENCE_ONLY)
    include("${CMAKE_CURRENT_SOURCE_DIR}/build-support/ReferenceBuild.cmake")
    return()
endif()
project(CP2077CoopSession VERSION 0.1.0 LANGUAGES CXX)

option(COOP_BUILD_PLUGIN "Build the Windows RED4ext session plugin" OFF)
option(COOP_BUILD_LEGACY_SERVER "Build the frozen Windows relay" ${WIN32})
option(COOP_SANITIZE "Enable Address/Undefined sanitizers on GCC/Clang" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(COOP_BUILD_SESSION_SERVER "Build portable session/state/relay service" ON)
include(CTest)
add_subdirectory(shared)
if(COOP_BUILD_SESSION_SERVER)
    add_subdirectory(SessionServer)
endif()
if(COOP_BUILD_PLUGIN)
    if(NOT WIN32)
        message(FATAL_ERROR "The RED4ext plugin requires Windows")
    endif()
    add_subdirectory(CoopPlugin)
endif()
if(COOP_BUILD_LEGACY_SERVER)
    if(NOT WIN32)
        message(FATAL_ERROR "Legacy relay requires Windows; the portable session core is built separately")
    endif()
    add_subdirectory(CoopServer)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
